import { useEffect, useState } from "react";
import { useNavigate, NavLink, useLocation, Outlet } from "react-router-dom";
import styles from "./gestion-moldes.module.css";

type Molde = {
  idMolde: number;
  nombre: string;
  descripcion: string | null;
  numeroCamadas: number;
  numeroEnvases: number;
  numeroGruesas: number;
};

const API_MOLDES = "http://127.0.0.1:8000/api/gestion-moldes";
const API_EXPORTAR_MOLDES = "http://127.0.0.1:8000/api/exportacion/moldes";

export const GestionMoldes = () => {
  const navigate = useNavigate();
  const location = useLocation();
  const [moldes, setMoldes] = useState<Molde[]>([]);
  const [cargando, setCargando] = useState(true);
  const [error, setError] = useState("");
  const [errorDescarga, setErrorDescarga] = useState("");
  const [descargando, setDescargando] = useState(false);

  const cargarMoldes = async () => {
    setCargando(true);
    setError("");
    try {
      const response = await fetch(API_MOLDES);
      if (!response.ok) {
        throw new Error(`Error al cargar moldes: ${response.status}`);
      }
      const data = await response.json();
      const moldesNormalizados: Molde[] = data.map((molde: Molde) => ({
        ...molde,
        idMolde: Number(molde.idMolde),
        numeroCamadas: Number(molde.numeroCamadas),
        numeroEnvases: Number(molde.numeroEnvases),
        numeroGruesas: Number(molde.numeroGruesas),
      }));
      // Ordenar los moldes por ID de forma ascendente
      const moldesOrdenados = moldesNormalizados.sort(
        (a, b) => a.idMolde - b.idMolde,
      );
      setMoldes(moldesOrdenados);
    } catch (loadError) {
      setError(
        loadError instanceof Error
          ? loadError.message
          : "Error al cargar moldes.",
      );
    } finally {
      setCargando(false);
    }
  };

  // Cargar moldes al montar el componente
  useEffect(() => {
    cargarMoldes();
  }, []);

  // Recargar cuando la ruta cambie a la raíz (cuando volvemos de registrar/editar)
  useEffect(() => {
    if (location.pathname === "/gestion-moldes") {
      cargarMoldes();
    }
  }, [location.pathname]);

  const handleDescargarExcel = async () => {
    setDescargando(true);
    setErrorDescarga("");
    try {
      const response = await fetch(API_EXPORTAR_MOLDES, {
        method: "POST",
        body: "",
      });

      if (!response.ok) {
        // Intentar leer el error como JSON, y si falla, usar un mensaje genérico
        let mensajeError = `Error al generar el Excel: ${response.status}`;
        try {
          const data = await response.json();
          if (data?.detail) {
            mensajeError =
              typeof data.detail === "string"
                ? data.detail
                : JSON.stringify(data.detail);
          }
        } catch {
          // La respuesta no era JSON, dejamos el mensaje genérico
        }
        throw new Error(mensajeError);
      }

      // Obtener el nombre del archivo desde Content-Disposition (si existe), sino se utiliza el nombre establecido:
      const contentDisposition = response.headers.get("Content-Disposition");
      let nombreArchivo = "Listado de Moldes.xlsx";
      if (contentDisposition) {
        const match = contentDisposition.match(/filename="?([^"]+)"?/);
        if (match && match[1]) {
          nombreArchivo = match[1];
        }
      }

      // Convertir la respuesta a Blob y descargar
      const blob = await response.blob();
      const url = window.URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.download = nombreArchivo;
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
      window.URL.revokeObjectURL(url);
    } catch (downloadError) {
      setErrorDescarga(
        downloadError instanceof Error
          ? downloadError.message
          : "Error al descargar el Excel.",
      );
    } finally {
      setDescargando(false);
    }
  };

  const isRoot = location.pathname === "/gestion-moldes";

  const getTitle = () => {
    const path = location.pathname;
    if (path.includes("registrar-molde")) return "Registrar Molde";
    if (path.includes("editar-molde")) return "Editar Molde";
    return "";
  };

  return (
    <section className="container">
      <header className={styles.moduleHeader}>
        <h2 className={styles.moduleTitle}>Gestión de Moldes</h2>
        <p className={styles.moduleSubtitle}>
          Catálogo de los moldes disponibles en la aplicación
        </p>
      </header>

      {!isRoot ? (
        <nav className={`${styles.subNav} content`} id="subNav">
          <NavLink
            to="/gestion-moldes"
            className={({ isActive }) =>
              isActive
                ? `${styles.subNavLink} ${styles.active}`
                : styles.subNavLink
            }
          >
            ← Volver
          </NavLink>
          <span className={styles.subNavTitle}>{getTitle()}</span>
        </nav>
      ) : null}

      {isRoot ? (
        <section className={`${styles.gestionMoldes} content`}>
          {/* === PANEL DE CONTROL === */}
          <section className={styles.controlPanel}>
            <h2 className={styles.title}>Listado de Moldes</h2>

            <div className={styles.buttonWrapper}>
              <button
                className={styles.downloadButton}
                onClick={handleDescargarExcel}
                disabled={descargando || cargando}
              >
                {descargando ? "Descargando..." : "↓ Descargar"}
              </button>
              <button
                className={styles.registerButton}
                onClick={() => navigate("registrar-molde")}
              >
                + Registrar Molde
              </button>
            </div>
          </section>

          {errorDescarga && <p className={styles.error}>{errorDescarga}</p>}

          {cargando && <p className={styles.loading}>Cargando moldes...</p>}
          {error && <p className={styles.error}>{error}</p>}
          {!cargando && !error && (
            <div className={styles.tableContainer}>
              {moldes.length === 0 ? (
                <p className={styles.emptyMessage}>
                  No hay moldes registrados.
                </p>
              ) : (
                <table className={styles.table}>
                  <thead>
                    <tr>
                      <th>Nombre</th>
                      <th>Descripción</th>
                      <th>Camadas</th>
                      <th>Envases</th>
                      <th>Gruesas</th>
                    </tr>
                  </thead>
                  <tbody>
                    {moldes.map((molde) => (
                      <tr
                        key={molde.idMolde}
                        onClick={() =>
                          navigate(`editar-molde/${molde.idMolde}`)
                        }
                        className={styles.clickableRow}
                      >
                        <td>{molde.nombre}</td>
                        <td>{molde.descripcion || "—"}</td>
                        <td>{molde.numeroCamadas}</td>
                        <td>{molde.numeroEnvases}</td>
                        <td>{molde.numeroGruesas}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              )}
            </div>
          )}
        </section>
      ) : (
        <Outlet />
      )}
    </section>
  );
};

Este es GestionMoldes/RegistrarMolde/index.tsx:

import { useState, type FormEvent } from "react";
import { useNavigate } from "react-router-dom";
import { HeaderForm } from "../../../components/HeaderForm";
import { ENVASES_POR_GRUESA } from "../../../../constants";
import styles from "./registrar-molde.module.css";

type FormData = {
  nombre: string;
  descripcion: string;
  numeroCamadas: number;
  numeroEnvases: number;
  numeroGruesas: number;
};

const API_MOLDES = "http://127.0.0.1:8000/api/gestion-moldes";

export const RegistrarMolde = () => {
  const navigate = useNavigate();
  const [formData, setFormData] = useState<FormData>({
    nombre: "",
    descripcion: "",
    numeroCamadas: 0,
    numeroEnvases: 0,
    numeroGruesas: 0,
  });
  const [error, setError] = useState("");
  const [cargando, setCargando] = useState(false);

  const handleChange = (
    e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>,
  ) => {
    const { name, value, type } = e.target;

    if (name === "numeroEnvases") {
      const envases = value === "" ? 0 : Number(value);
      const gruesas = envases / ENVASES_POR_GRUESA;
      setFormData((prev) => ({
        ...prev,
        numeroEnvases: envases,
        numeroGruesas: Number(gruesas.toFixed(2)),
      }));
    } else {
      setFormData((prev) => ({
        ...prev,
        [name]: type === "number" ? (value === "" ? 0 : Number(value)) : value,
      }));
    }
  };

  const handleSubmit = async (event: FormEvent<HTMLFormElement>) => {
    event.preventDefault();
    setError("");
    setCargando(true);

    // Validaciones
    if (!formData.nombre.trim()) {
      setError("El nombre del molde es obligatorio.");
      setCargando(false);
      return;
    }
    if (formData.numeroCamadas <= 0) {
      setError("El número de camadas debe ser mayor a 0.");
      setCargando(false);
      return;
    }
    if (formData.numeroEnvases <= 0) {
      setError("El número de envases debe ser mayor a 0.");
      setCargando(false);
      return;
    }

    // Preparar datos para enviar (incluyendo numeroGruesas calculado)
    const datosEnviar = {
      ...formData,
      numeroGruesas: formData.numeroEnvases / ENVASES_POR_GRUESA,
    };

    try {
      const response = await fetch(API_MOLDES, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(datosEnviar),
      });

      if (!response.ok) {
        const data = await response.json().catch(() => null);
        throw new Error(
          data?.detail ?? `Error al registrar molde: ${response.status}`,
        );
      }

      navigate("/gestion-moldes");
    } catch (submitError) {
      setError(
        submitError instanceof Error
          ? submitError.message
          : "Error al registrar molde.",
      );
    } finally {
      setCargando(false);
    }
  };

  const handleReset = () => {
    setFormData({
      nombre: "",
      descripcion: "",
      numeroCamadas: 0,
      numeroEnvases: 0,
      numeroGruesas: 0,
    });
    setError("");
  };

  return (
    <form className={`${styles.form} content`} onSubmit={handleSubmit}>
      <HeaderForm />

      <h3 className={styles.title}>Registrar Nuevo Molde</h3>

      <section className={styles.formData}>
        <fieldset className={styles.fieldset}>
          <legend className={styles.legend}>Datos del Molde</legend>

          <div className={styles.inputWrapper}>
            <label htmlFor="nombre">
              Nombre del Molde <span className={styles.requireSymbol}>*</span>
            </label>
            <input
              id="nombre"
              name="nombre"
              type="text"
              value={formData.nombre}
              onChange={handleChange}
              required
            />
          </div>

          <div className={styles.inputWrapper}>
            <label htmlFor="descripcion">
              Descripción <span className={styles.requireSymbol}>*</span>
            </label>
            <textarea
              id="descripcion"
              name="descripcion"
              value={formData.descripcion}
              onChange={handleChange}
              rows={6}
              required
            />
          </div>

          <div className={styles.numbers}>
            <div className={styles.inputWrapper}>
              <label htmlFor="numeroCamadas">
                Número de Camadas{" "}
                <span className={styles.requireSymbol}>*</span>
              </label>
              <input
                id="numeroCamadas"
                name="numeroCamadas"
                type="number"
                min="1"
                value={formData.numeroCamadas || ""}
                onChange={handleChange}
                required
              />
            </div>

            <div className={styles.inputWrapper}>
              <label htmlFor="numeroEnvases">
                Número de Envases{" "}
                <span className={styles.requireSymbol}>*</span>
              </label>
              <input
                id="numeroEnvases"
                name="numeroEnvases"
                type="number"
                min="1"
                value={formData.numeroEnvases || ""}
                onChange={handleChange}
                required
              />
            </div>

            <div className={styles.inputWrapper}>
              <label htmlFor="numeroGruesas">
                Número de Gruesas{" "}
                <span className={styles.requireSymbol}>*</span>
              </label>
              <input
                id="numeroGruesas"
                name="numeroGruesas"
                type="text"
                value={formData.numeroGruesas || ""}
                readOnly
                className={styles.readonlyInput}
              />
            </div>
          </div>
        </fieldset>

        {error && <p className={styles.error}>{error}</p>}

        <div className={styles.buttonWrapper}>
          <button
            type="submit"
            className={styles.saveButton}
            disabled={cargando}
          >
            {cargando ? "Registrando..." : "Guardar"}
          </button>
          <button
            type="button"
            className={styles.resetButton}
            onClick={handleReset}
            disabled={cargando}
          >
            Resetear
          </button>
        </div>
      </section>
    </form>
  );
};

    numeroGruesas: 0,

  const [error, setError] = useState("");
  const [guardando, setGuardando] = useState(false);

  useEffect(() => {
    const cargarMolde = async () => {
      setCargando(true);
      setError("");
      try {
        const response = await fetch(`${API_MOLDES}/${idMolde}`);
        if (!response.ok) {
          const data = await response.json().catch(() => null);
          throw new Error(
            data?.detail ??
              (response.status === 404
                ? "El molde especificado no existe."
                : `Error al cargar molde: ${response.status}`),
          );
        }
        const molde = (await response.json()) as Molde;
        setFormData({
          nombre: molde.nombre,
          descripcion: molde.descripcion || "",
          numeroCamadas: Number(molde.numeroCamadas),
          numeroEnvases: Number(molde.numeroEnvases),
          numeroGruesas: Number(
            (Number(molde.numeroEnvases) / ENVASES_POR_GRUESA).toFixed(2),
          ),
        });
      } catch (loadError) {
        setError(
          loadError instanceof Error
            ? loadError.message
            : "Error al cargar molde.",
        );
      } finally {
        setCargando(false);
      }
    };
    void cargarMolde();
  }, [idMolde]);

  const handleChange = (
    e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>,
  ) => {
    const { name, value, type } = e.target;

    if (name === "numeroEnvases") {
      const envases = value === "" ? 0 : Number(value);
      const gruesas = envases / ENVASES_POR_GRUESA;
      setFormData((prev) => ({
        ...prev,
        numeroEnvases: envases,
        numeroGruesas: Number(gruesas.toFixed(2)),
      }));
    } else {
      setFormData((prev) => ({
        ...prev,
        [name]: type === "number" ? (value === "" ? 0 : Number(value)) : value,
      }));
    }
  };

  const handleSubmit = async (event: FormEvent<HTMLFormElement>) => {
    event.preventDefault();
    setError("");
    setGuardando(true);

    // Validaciones
    if (!formData.nombre.trim()) {
      setError("El nombre del molde es obligatorio.");
      setGuardando(false);
      return;
    }
    if (formData.numeroCamadas <= 0) {
      setError("El número de camadas debe ser mayor a 0.");
      setGuardando(false);
      return;
    }
    if (formData.numeroEnvases <= 0) {
      setError("El número de envases debe ser mayor a 0.");
      setGuardando(false);
      return;
    }

    // Preparar datos para enviar
    const datosEnviar = {
      nombre: formData.nombre,
      descripcion: formData.descripcion,
      numeroCamadas: formData.numeroCamadas,
      numeroEnvases: formData.numeroEnvases,
      numeroGruesas: formData.numeroEnvases / ENVASES_POR_GRUESA,
    };

    try {
      const response = await fetch(`${API_MOLDES}/${idMolde}`, {
        method: "PUT",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(datosEnviar),
      });

      if (!response.ok) {
        const data = await response.json().catch(() => null);
        throw new Error(
          data?.detail ?? `Error al actualizar molde: ${response.status}`,
        );
      }

      navigate("/gestion-moldes");
    } catch (submitError) {
      setError(
        submitError instanceof Error
          ? submitError.message
          : "Error al actualizar molde.",
      );
    } finally {
      setGuardando(false);
    }
  };

  const handleReset = () => {
    const cargarMolde = async () => {
      try {
        const response = await fetch(`${API_MOLDES}/${idMolde}`);
        if (response.ok) {
          const molde = (await response.json()) as Molde;
          setFormData({
            nombre: molde.nombre,
            descripcion: molde.descripcion || "",
            numeroCamadas: Number(molde.numeroCamadas),
            numeroEnvases: Number(molde.numeroEnvases),
            numeroGruesas: Number(
              (Number(molde.numeroEnvases) / ENVASES_POR_GRUESA).toFixed(2),
            ),
          });
        }
      } catch {
        // Ignorar errores en reset
      }
    };
    void cargarMolde();
    setError("");
  };

  if (cargando) {
    return <p className={styles.loading}>Cargando molde...</p>;
  }

  return (
    <form className={`${styles.form} content`} onSubmit={handleSubmit}>
      <HeaderForm />

      <h3 className={styles.title}>Editar Molde #{idMolde}</h3>

      <section className={styles.formData}>
        <fieldset className={styles.fieldset}>
          <legend className={styles.legend}>Datos del Molde</legend>

          <div className={styles.inputWrapper}>
            <label htmlFor="nombre">
              Nombre del Molde <span className={styles.requireSymbol}>*</span>
            </label>
            <input
              id="nombre"
              name="nombre"
              type="text"
              value={formData.nombre}
              onChange={handleChange}
              placeholder="Ej: Molde A"
              required
            />
          </div>

          <div className={styles.inputWrapper}>
            <label htmlFor="descripcion">
              Descripción <span className={styles.requireSymbol}>*</span>
            </label>
            <textarea
              id="descripcion"
              name="descripcion"
              value={formData.descripcion}
              onChange={handleChange}
              placeholder="Descripción del molde"
              rows={6}
              required
            />
          </div>

          <div className={styles.numbers}>
            <div className={styles.inputWrapper}>
              <label htmlFor="numeroCamadas">
                Número de Camadas{" "}
                <span className={styles.requireSymbol}>*</span>
              </label>
              <input
                id="numeroCamadas"
                name="numeroCamadas"
                type="number"
                min="1"
                value={formData.numeroCamadas || ""}
                onChange={handleChange}
                required
              />
            </div>

            <div className={styles.inputWrapper}>
              <label htmlFor="numeroEnvases">
                Número de Envases{" "}
                <span className={styles.requireSymbol}>*</span>
              </label>
              <input
                id="numeroEnvases"
                name="numeroEnvases"
                type="number"
                min="1"
                value={formData.numeroEnvases || ""}
                onChange={handleChange}
                required
              />
            </div>

            <div className={styles.inputWrapper}>
              <label htmlFor="numeroGruesas">
                Número de Gruesas{" "}
                <span className={styles.requireSymbol}>*</span>
              </label>
              <input
                id="numeroGruesas"
                name="numeroGruesas"
                type="text"
                value={formData.numeroGruesas || ""}
                readOnly
                className={styles.readonlyInput}
              />
            </div>
          </div>
        </fieldset>

        {error && <p className={styles.error}>{error}</p>}

        <div className={styles.buttonWrapper}>
          <button
            type="submit"
            className={styles.saveButton}
            disabled={guardando}
          >
            {guardando ? "Actualizando..." : "Actualizar"}
          </button>
          <button
            type="button"
            className={styles.resetButton}
            onClick={handleReset}
            disabled={guardando}
          >
            Resetear
          </button>
        </div>
      </section>
    </form>
  );
};
