import { useNavigate } from "react-router-dom";
import { HeaderForm } from "../../../components/HeaderForm";
import { ENVASES_POR_GRUESA } from "../../../../constants";
import styles from "./registrar-molde.module.css";

type FormData = {
  nombre: string;
  descripcion: string;
  numeroCamadas: number;
  numeroEnvases: number;
  numeroGruesas: number;
};

const API_MOLDES = "http://127.0.0.1:8000/api/gestion-moldes";

export const RegistrarMolde = () => {
  const navigate = useNavigate();
  const [formData, setFormData] = useState<FormData>({
    nombre: "",
    descripcion: "",
    numeroCamadas: 0,
    numeroEnvases: 0,
    numeroGruesas: 0,
  });
  const [error, setError] = useState("");
  const [cargando, setCargando] = useState(false);

  const handleChange = (
    e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>,
  ) => {
    const { name, value, type } = e.target;

    if (name === "numeroEnvases") {
      const envases = value === "" ? 0 : Number(value);
      const gruesas = envases / ENVASES_POR_GRUESA;
      setFormData((prev) => ({
        ...prev,
        numeroEnvases: envases,
        numeroGruesas: Number(gruesas.toFixed(2)),
      }));
    } else {
      setFormData((prev) => ({
        ...prev,
        [name]: type === "number" ? (value === "" ? 0 : Number(value)) : value,
      }));
    }
  };

  const handleSubmit = async (event: FormEvent<HTMLFormElement>) => {
    event.preventDefault();
    setError("");
    setCargando(true);

    // Validaciones
    if (!formData.nombre.trim()) {
      setError("El nombre del molde es obligatorio.");
      setCargando(false);
      return;
    }
    if (formData.numeroCamadas <= 0) {
      setError("El número de camadas debe ser mayor a 0.");
      setCargando(false);
      return;
    }
    if (formData.numeroEnvases <= 0) {
      setError("El número de envases debe ser mayor a 0.");
      setCargando(false);
      return;
    }

    // Preparar datos para enviar (incluyendo numeroGruesas calculado)
    const datosEnviar = {
      ...formData,
      numeroGruesas: formData.numeroEnvases / ENVASES_POR_GRUESA,
    };

    try {
      const response = await fetch(API_MOLDES, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(datosEnviar),
      });

      if (!response.ok) {
        const data = await response.json().catch(() => null);
        throw new Error(
          data?.detail ?? `Error al registrar molde: ${response.status}`,
        );
      }

      navigate("/gestion-moldes");
    } catch (submitError) {
      setError(
        submitError instanceof Error
          ? submitError.message
          : "Error al registrar molde.",
      );
    } finally {
      setCargando(false);
    }
  };

    setFormData({
      nombre: "",
      descripcion: "",
      numeroCamadas: 0,
      numeroEnvases: 0,
      numeroGruesas: 0,
    });
    setError("");
  };

  return (
    <form className={`${styles.form} content`} onSubmit={handleSubmit}>
      <HeaderForm />

            </label>
            <input
              id="nombre"
              name="nombre"
              type="text"
              value={formData.nombre}
              onChange={handleChange}
              required
            />
          </div>

          <div className={styles.inputWrapper}>
            <label htmlFor="descripcion">
              Descripción <span className={styles.requireSymbol}>*</span>
            </label>
            <textarea
              id="descripcion"
              name="descripcion"
              value={formData.descripcion}
              onChange={handleChange}
              rows={6}
              required
            />
          </div>

          <div className={styles.numbers}>
            <div className={styles.inputWrapper}>
              <label htmlFor="numeroCamadas">
                Número de Camadas{" "}
                <span className={styles.requireSymbol}>*</span>
              </label>
              <input
                id="numeroCamadas"
                name="numeroCamadas"
                type="number"
                min="1"
                value={formData.numeroCamadas || ""}
                onChange={handleChange}
                required
              />
            </div>

            <div className={styles.inputWrapper}>
              <label htmlFor="numeroEnvases">
                Número de Envases{" "}
                <span className={styles.requireSymbol}>*</span>
              </label>
              <input
                id="numeroEnvases"
                name="numeroEnvases"
                type="number"
                min="1"
                value={formData.numeroEnvases || ""}

                required

                <span className={styles.requireSymbol}>*</span>

              <input
                id="numeroGruesas"
                name="numeroGruesas"
                type="text"
                value={formData.numeroGruesas || ""}
                readOnly

            </div>

        </fieldset>

  );

        if (!response.ok) {
          const data = await response.json().catch(() => null);
          throw new Error(
            data?.detail ??
              (response.status === 404
                ? "El molde especificado no existe."
                : `Error al cargar molde: ${response.status}`),
          );

          descripcion: molde.descripcion || "",
          numeroCamadas: Number(molde.numeroCamadas),
          numeroEnvases: Number(molde.numeroEnvases),

// Tipos genéricos

  id: string;

  carnet: string;
  nombres: string;
  apellidos: string;

  id_permisom: number;

  accion: string;
};

  data: {
    usuario: {

      carnet_usuario: string;

      apellidos: string;

  usuarioId: string | null;
  onGuardar: (usuario: any) => void; // Cambiamos a any para flexibilidad
  onCancelar: () => void;
}

  cedula: string;
  carnet: string;
  nombres: string;
  apellidos: string;
  telefono: string;
  correo: string;
  rol: string;
  estado: "Activo" | "Inactivo";

  usuarioId,
  onGuardar,
  onCancelar,
}: FormularioUsuarioProps) => {
  const [formData, setFormData] = useState<FormularioData>({
    cedula: "",
    carnet: "",
    nombres: "",
    apellidos: "",
    telefono: "",
    correo: "",
    rol: "",
    estado: "Activo",
    password: "",
    permisosSeleccionados: [],
  });

  const [catalogoPermisos, setCatalogoPermisos] = useState<PermisoCatalogo[]>(
    [],
  );
  const [cargandoPermisos, setCargandoPermisos] = useState(false);
  const [cargandoUsuario, setCargandoUsuario] = useState(false);
  const esEdicion = usuarioId !== null;

  //* Obtener el catálogo de permisos
  const obtenerCatalogoPermisos = useCallback(async () => {
    try {
      setCargandoPermisos(true);
      const response = await fetch(
        "http://127.0.0.1:8000/api/usuarios/permisos/catalogo",
      );
      if (!response.ok)
        throw new Error("Error al obtener el catálogo de permisos");
      const data = await response.json();

      // Extraer el array de permisos sin importar la estructura
      let permisosArray = [];

      if (data.data) {
        if (Array.isArray(data.data)) {
          permisosArray = data.data;
        } else if (data.data.data && Array.isArray(data.data.data)) {
          permisosArray = data.data.data;
        } else {
          permisosArray = [];
        }
      } else if (Array.isArray(data)) {
        permisosArray = data;
      }

      setCatalogoPermisos(permisosArray);
    } catch (err) {
      console.error("Error al obtener catálogo de permisos:", err);
      alert("Error al cargar los permisos disponibles");
      setCatalogoPermisos([]);
    } finally {
      setCargandoPermisos(false);
    }
  }, []);

  //* Obtener datos del usuario por ID (para edición)
  const obtenerUsuarioPorId = useCallback(
    async (id: string) => {
      try {
        setCargandoUsuario(true);
        const response = await fetch(
          `http://127.0.0.1:8000/api/usuarios/${id}`,
        );
        if (!response.ok) {
          if (response.status === 404) {

          }

        // Cargar los datos en el formulario

                    handlePermisoChange(permiso.id_permisom, e.target.checked)

        <div className={styles.loadingContainer}>
          <p>Cargando datos del usuario...</p>

  return (
